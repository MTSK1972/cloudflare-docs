  <script>
    // *** เปลี่ยน URL นี้ตอน deploy จริง ***
    const BACKEND_URL = window.location.hostname === 'localhost' 
      ? 'http://localhost:8000'
      : 'https://api.moaisu.org';
    
    let currentLang = 'th';
    let recognition = null;
    let isListening = false;
    let isProcessing = false;
    
    const LANG_CONFIG = {
      th: { code: 'th-TH', name: 'Thai',     placeholder: 'พิมพ์หรือกดไมค์ถาม...' },
      en: { code: 'en-US', name: 'English',  placeholder: 'Ask about Dhamma...' },
      zh: { code: 'zh-CN', name: 'Chinese',  placeholder: '询问佛法...' },
      ja: { code: 'ja-JP', name: 'Japanese', placeholder: '仏法について...' }
    };
    
    // ----- Init -----
    window.addEventListener('DOMContentLoaded', () => {
      checkBackend();
      setupSpeechRecognition();
      setInterval(checkBackend, 30000);
    });
    
    async function checkBackend() {
      try {
        const res = await fetch(BACKEND_URL + '/api/info', { 
          mode: 'cors',
          signal: AbortSignal.timeout(5000)
        });
        if (res.ok) {
          document.getElementById('statusDot').className = 'status-dot online';
          document.getElementById('statusText').textContent = 'พร้อม';
        } else throw new Error();
      } catch (e) {
        document.getElementById('statusDot').className = 'status-dot offline';
        document.getElementById('statusText').textContent = 'ออฟไลน์';
      }
    }
    
    function selectLang(lang, btn) {
      currentLang = lang;
      document.querySelectorAll('.lang-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      document.getElementById('textInput').placeholder = LANG_CONFIG[lang].placeholder;
      if (recognition) recognition.lang = LANG_CONFIG[lang].code;
    }
    
    function setupSpeechRecognition() {
      const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SR) return;
      recognition = new SR();
      recognition.continuous = false;
      recognition.interimResults = false;
      recognition.lang = LANG_CONFIG[currentLang].code;
      
      recognition.onstart = () => {
        isListening = true;
        document.getElementById('micBtn').classList.add('listening');
        document.getElementById('voiceWave').classList.add('active');
      };
      
      recognition.onresult = (e) => {
        const transcript = e.results[0][0].transcript;
        document.getElementById('textInput').value = transcript;
        setTimeout(() => handleSubmit(), 300);
      };
      
      recognition.onerror = () => stopListening();
      recognition.onend = () => stopListening();
    }
    
    function toggleVoice() {
      if (!recognition) {
        alert('ใช้ Chrome หรือ Edge เพื่อสั่งงานด้วยเสียง');
        return;
      }
      if (isListening) recognition.stop();
      else {
        recognition.lang = LANG_CONFIG[currentLang].code;
        try { recognition.start(); } catch(e) {}
      }
    }
    
    function stopListening() {
      isListening = false;
      document.getElementById('micBtn').classList.remove('listening');
      document.getElementById('voiceWave').classList.remove('active');
    }
    
    function askQuestion(q) {
      document.getElementById('textInput').value = q;
      handleSubmit();
    }
    
    async function handleSubmit() {
      if (isProcessing) return;
      const input = document.getElementById('textInput');
      const question = input.value.trim();
      if (!question) return;
      
      isProcessing = true;
      input.value = '';
      
      addMessage(question, 'user');
      
      const loadingDiv = document.createElement('div');
      loadingDiv.className = 'chat-msg bot typing';
      loadingDiv.textContent = 'กำลังหาคำตอบ...';
      document.getElementById('console').appendChild(loadingDiv);
      scrollConsole();
      
      try {
        const res = await fetch(BACKEND_URL + '/api/chat', {
          method: 'POST',
          mode: 'cors',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ message: question, lang: currentLang }),
          signal: AbortSignal.timeout(60000)
        });
        
        if (!res.ok) throw new Error('Backend error: ' + res.status);
        const data = await res.json();
        
        loadingDiv.classList.remove('typing');
        loadingDiv.innerHTML = data.reply || '...';
        
        // แสดงแหล่งที่มา (ถ้ามี)
        if (data.sources && data.sources.length > 0) {
          const sourcesDiv = document.createElement('div');
          sourcesDiv.style.marginTop = '6px';
          data.sources.forEach(src => {
            const tag = document.createElement('a');
            tag.className = 'source-tag';
            tag.textContent = '🪷 ' + src.title;
            tag.href = src.url || '#';
            tag.target = '_blank';
            sourcesDiv.appendChild(tag);
          });
          loadingDiv.appendChild(sourcesDiv);
        }
        
        // พูดออกเสียง (เสียงมาตรฐาน)
        speak(data.reply, currentLang);
        
      } catch (e) {
        loadingDiv.classList.remove('typing');
        loadingDiv.innerHTML = '⚠️ ระบบยังไม่พร้อม — กรุณาตรวจสอบ Backend (' + e.message + ')';
      }
      
      isProcessing = false;
      scrollConsole();
    }
    
    function addMessage(text, type) {
      const div = document.createElement('div');
      div.className = 'chat-msg ' + type;
      div.textContent = text;
      document.getElementById('console').appendChild(div);
      scrollConsole();
    }
    
    function scrollConsole() {
      const c = document.getElementById('console');
      c.scrollTop = c.scrollHeight;
    }
    
    // ----- TTS (เสียงมาตรฐาน) -----
    function speak(text, lang) {
      if (!text || !window.speechSynthesis) return;
      window.speechSynthesis.cancel();
      
      // ลบ HTML tags ออกก่อนพูด
      const cleanText = text.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ').trim();
      
      const utter = new SpeechSynthesisUtterance(cleanText);
      utter.lang = LANG_CONFIG[lang].code;
      utter.rate = 0.95;
      utter.pitch = 1.0;
      utter.volume = 1.0;
      
      const card = document.getElementById('avatarCard');
      const wave = document.getElementById('voiceWave');
      
      utter.onstart = () => {
        card.classList.add('speaking');
        wave.classList.add('active');
      };
      utter.onend = () => {
        card.classList.remove('speaking');
        wave.classList.remove('active');
      };
      
      window.speechSynthesis.speak(utter);
    }
